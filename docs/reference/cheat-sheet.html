Djazair Syntax Cheat Sheet

A quick-reference guide summarizing syntax, control flow, functions, OOP, and common idioms.

Basics & Variables

djazair 
let x = 10                  # Integer
let pi = 3.14               # Float
let str = "Hello ${x}"      # String interpolation
let multiline = `row 1
row 2`
let arr = [1, 2, 3]         # Array
let map = {"key": "val"}    # Map
let isOk = True             # Bool
let empty = Null            # Null

Functions

djazair 
# Standard
fn add(a, b = 1)
    return a + b
end

# Arrow expression
let square = fn(x) => x * x

# Rest params
fn logAll(...items)
    print(items)
end

Control Structures

djazair 
let x = 10

# If / Elif / Else
if x > 10
    print("Greater")
elif x == 10
    print("Equal")
else
    print("Smaller")
end

# Match-case
match x
    case 1, 2 print("Few")
    case 10   print("Ten")
    default   print("Other")
end

# For-in loop
for item in [1, 2, 3] print(item) end
for k, v in {"a": 1} print("${k}: ${v}") end
for i in 0..5 print(i) end

Classes & OOP

djazair 
class Animal
    init(name) self.name = name end
    speak() return "Roar" end
end

class Cat is Animal
    init(name, breed)
        super.init(name)
        self.breed = breed
    end
    speak() return "Meow" end
end

let c = new Cat("Luna", "Siamese")
print(c.speak()) # => Meow
print(c instanceof Animal) # => True
